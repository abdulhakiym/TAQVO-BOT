# --- OBUNA TEKSHIRISH (ENG MUHIMI) ---
async def check_sub(update: Update, context: ContextTypes.DEFAULT_TYPE):
    try:
        user_id = update.effective_user.id
        member = await context.bot.get_chat_member(f"@{CHANNEL_USERNAME}", user_id)
        if member.status in ['member', 'administrator', 'creator']:
            return True
        else:
            raise Exception("Not sub")
    except:
        keyboard = [[InlineKeyboardButton("📢 Kanalga obuna bo'lish", url=CHANNEL_LINK)],
                    [InlineKeyboardButton("✅ Obuna bo'ldim, tekshir", callback_data="check_sub")]]
        await update.message.reply_text(
            f"⛔ Botdan foydalanish uchun kanalimizga obuna bo'lishingiz shart!\n\n👉 @{CHANNEL_USERNAME} ga obuna bo'ling, keyin 'Tekshir' ni bosing.",
            reply_markup=InlineKeyboardMarkup(keyboard)
        )
        return False

async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if not await check_sub(update, context): return
    keyboard = [
        ["🕌 Namoz vaqtlari", "📿 Tasbeh"],
        ["🌅 Tonggi zikrlar", "🌙 Kechki zikrlar"],
        ["📖 Qur'on o'qish", "📜 Yodlash uchun hadis"],
        ["🧭 Qibla", "📅 Hijriy sana"]
    ]
    await update.message.reply_text(f"Assalomu alaykum {update.effective_user.first_name}!\nTAQVO-BOT ga xush kelibsiz 🤲\nKaraganda vaqti bilan ishlayman.", reply_markup=ReplyKeyboardMarkup(keyboard, resize_keyboard=True))

async def namoz_vaqti_func():
    try:
        url = f"http://api.aladhan.com/v1/timingsByCity?city=Karaganda&country=Kazakhstan&method=2"
        r = requests.get(url).json()
        return r['data']['timings']
    except: return None

async def send_namoz(context: ContextTypes.DEFAULT_TYPE):

# QUR'ON - Bot ichida o'qish
async def quran_start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if not await check_sub(update, context): return
    keyboard = [[InlineKeyboardButton("Fotiha", callback_data="surah_1"), InlineKeyboardButton("Ixlos", callback_data="surah_112")],
                [InlineKeyboardButton("Baqara", callback_data="surah_2"), InlineKeyboardButton("Yosin", callback_data="surah_36")]]
    await update.message.reply_text("📖 Qur'on - Surani tanlang:", reply_markup=InlineKeyboardMarkup(keyboard))

async def handle_callback(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()
    if query.data.startswith("surah_"):
        surah_id = query.data.split("_")[1]
        try:
            r = requests.get(f"https://api.alquran.cloud/v1/surah/{surah_id}/uz.mansour").json()
            ayahs = r['data']['ayahs'][:5] # 5 ta oyat namuna
            text = f"📖 {r['data']['name']} surasi:\n\n"
            for a in ayahs:
                text += f"{a['numberInSurah']}. {a['text']}\n\n"
            text += f"...davomi botda o'qiladi. To'liq o'qish uchun {surah_id}-sura"
            await query.edit_message_text(text)
        except:
            await query.edit_message_text("Hozircha yuklab bo'lmadi.")
    elif query.data == "check_sub":
        if await check_sub(query, context):
            await query.edit_message_text("✅ Rahmat! Obuna tasdiqlandi. /start bosing.")

# Qolgan funksiyalar
async def handle_text(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if not await check_sub(update, context): return
    txt = update.message.text
    if "Namoz" in txt:
        t = await namoz_vaqti_func()
        if t: await update.message.reply_text(f"🕌 Karaganda:\nBomdod {t['Fajr']}\nPeshin {t['Dhuhr']}\nAsr {t['Asr']}\nShom {t['Maghrib']}\nXufton {t['Isha']}")
    elif "Tasbeh" in txt:
        uid = update.effective_user.id
        tasbeh_counts[uid] = tasbeh_counts.get(uid, 0)+1
        await update.message.reply_text(f"📿 {tasbeh_counts[uid]} - SubhanAlloh!")
    elif "Tonggi" in txt:
        await update.message.reply_text("🌅 Tonggi zikrlar:\n\n" + "\n\n".join(MORNING_ADHKAR))
    elif "Kechki" in txt:
        await update.message.reply_text("🌙 Kechki zikrlar:\n\n" + "\n\n".join(EVENING_ADHKAR))
    elif "hadis" in txt.lower():
        await update.message.reply_text("📜 Yodlash uchun:\n\n" + random.choice(HADIS_40))
    elif "Qur'on" in txt:
        await quran_start(update, context)
    elif "Qibla" in txt:
        await update.message.reply_text("🧭 Qibla Karaganda uchun 215° Janubi-G'arb.")

async def akhira_job(context: ContextTypes.DEFAULT_TYPE):
    text = random.choice(AKHIRAT_ESLATMA)
    print(f"Eslatma yuborish: {text}")
    # Bu yerda barcha obunachilarga yuborish logikasi bo'ladi

def main():
    app = Application.builder().token(BOT_TOKEN).build()
    app.add_handler(CommandHandler("start", start))
    app.add_handler(CommandHandler("quran", quran_start))
    app.add_handler(CallbackQueryHandler(handle_callback))
    app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, handle_text))

    # Eslatmalar jadvali - Karaganda vaqti Asia/Almaty
    scheduler = AsyncIOScheduler(timezone=pytz.timezone("Asia/Almaty"))
    scheduler.add_job(lambda: app.create_task(send_namoz(app)), 'cron', hour=5, minute=0) # Har kuni 5:00 da namoz vaqtlari
    scheduler.add_job(lambda: app.create_task(akhira_job(app)), 'cron', hour=10, minute=0) # Oxirat eslatma
    scheduler.add_job(lambda: app.create_task(akhira_job(app)), 'cron', hour=21, minute=0) # Oxirat eslatma
    scheduler.start()

    print("TAQVO-BOT to'liq ishga tushdi...")
    app.run_polling()

if __name__ == "__main__":
    main()
